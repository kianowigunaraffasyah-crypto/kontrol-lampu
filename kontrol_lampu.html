import network import time from machine import Pin from umqtt.simple import MQTTClient # Lampu LED dan Saklar Fisik led = Pin(2, Pin.OUT) saklar_on = Pin(13, Pin.IN) saklar_off = Pin(21, Pin.IN) # Menghubungkan ke Wi-Fi wlan = network.WLAN(network.STA_IF) wlan.active(True) wlan.connect("jm", "123456789") while not wlan.isconnected(): time.sleep(0.5) # Menghubungkan ke Server Internet BROKER = "broker.hivemq.com" TOPIC = b"esp32/lampu/kontrol" def terima_sinyal(topic, msg): if msg == b"ON": led.value(1) elif msg == b"OFF": led.value(0) client = MQTTClient("esp32_papan_lampu", BROKER) client.set_callback(terima_sinyal) client.connect() client.subscribe(TOPIC) while True: # 1. Membaca Saklar Fisik if saklar_on.value() == 1: led.value(1) elif saklar_off.value() == 1: led.value(0) # 2. Membaca Tombol dari File HTML client.check_msg() time.sleep(0.05)
